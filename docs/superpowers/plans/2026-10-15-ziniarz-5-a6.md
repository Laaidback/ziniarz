# ZINIARZ, plan 5: format A6 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Switch ZINIARZ from A5 pages to A6 pages (105×148 mm): one A5 leaf printed at 100% in the middle of an A4 landscape sheet, cut out along the crop marks and folded; old A5 zines are converted to A6 when opened.

**Architecture:** The page size lives in `PAGE_WIDTH`/`PAGE_HEIGHT` (`src/model/geometry.ts`); renderer, editor, snapping, insertion and 3D follow them. A new pure module `src/model/migrate.ts` converts raw format-1 (A5) data to A6 mm before `parseZine` reads it (`FORMAT_VERSION` 2). Printing drops the printer margin and the scale: `printLayout` always places the leaf at 100%. A handful of A5-specific numbers (test page, sample zine, thumbnails, panel limit, CSS ratios) move to A6.

**Tech Stack:** Vite 7, React 19, TypeScript, Konva, Zustand, Vitest 3.2 (unit in Node, browser in Chromium), Playwright, pdf-lib.

**Spec:** `docs/superpowers/specs/2026-10-15-ziniarz-a6-design.md` (amends `docs/superpowers/specs/2026-10-14-ziniarz-design.md`). Read it before starting. Earlier plans: 1 core, 2 editor, 3 print and 3D, 4 hosting.

## Decisions

- **`PrintLayout.scale` is removed**, not kept as a constant 1. Its users (`sheetSide.ts`, `printTexts.ts`, tests) are few and the field would lie about a choice that no longer exists.
- **`PrintLayoutOptions`** becomes `Pick<PrintSettings, 'bleed' | 'cropMarks' | 'foldMarks'>`. Callers that pass a whole `PrintSettings` keep working.
- **Preview 3D** takes the page size from `PAGE_WIDTH`/`PAGE_HEIGHT` (the trimmed page is always the page now), not from `printLayout`.
- **Thumbnails:** `COVER_PX_PER_MM` 1.5 → 2 (cover canvas 210×296 px, whole pixels; slightly below the spec's ×148/105 ≈ 2.11, on purpose), `THUMB_PX_PER_MM` 0.5 → 0.7.
- **Factory fallbacks:** text width 60, photo 60×90.
- **Migration with zero-size frames:** `fx = kx`, `fy = ky` when the old width/height was 0 (spec says so).
- **The render-reference test fails from Task 2 until Task 6** (the sample zine and the print layout change on the way). That is expected; Task 6 rewrites the sample zine and refreshes the references on purpose. Every other test is green at the end of each task, except the ones a task names as fixed in a later task.
- **Nothing is pushed.** Pushing to `main` publishes the site; the owner decides after reviewing (final task ends with local commits).

```ts
import { BLEED } from './geometry';

/** Page of format version 1 (A5), mm. */
const A5 = { width: 148, height: 210 } as const;
/** Page of format version 2 (A6), mm. Equal to PAGE_WIDTH × PAGE_HEIGHT. */
const A6 = { width: 105, height: 148 } as const;
const KX = A6.width / A5.width;
const KY = A6.height / A5.height;

type Obj = Record<string, unknown>;
const isObj = (v: unknown): v is Obj => typeof v === 'object' && v !== null && !Array.isArray(v);
const isNum = (v: unknown): v is number => typeof v === 'number' && Number.isFinite(v);

/**
 * One axis from A5 to A6 mm: linear (0 → 0, old edge → new edge), also beyond the edges,
 * except that an overshoot into the bleed (up to BLEED) never shrinks, so bleeding photos
 * keep their full bleed.
 */
function mapAxis(v: number, oldEdge: number, newEdge: number): number {
  const k = newEdge / oldEdge;
  if (v < 0) return Math.min(v * k, Math.max(v, -BLEED));
  if (v > oldEdge) {
    const over = v - oldEdge;
    return Math.max(newEdge + over * k, Math.min(newEdge + over, newEdge + BLEED));
  }
  return v * k;
}

export const mapX = (x: number) => mapAxis(x, 2 * A5.width, 2 * A6.width);
export const mapY = (y: number) => mapAxis(y, A5.height, A6.height);

/** New position and size of one frame axis, plus the factor the size changed by. */
function mapSpan(start: number, size: number, map: (v: number) => number, fallback: number) {
  const from = map(start);
  const to = map(start + size);
  return { start: from, size: to - from, factor: size === 0 ? fallback : (to - from) / size };
}

function migrateElement(e: unknown): unknown {
  if (!isObj(e)) return e;
  const out: Obj = { ...e };
  let fx = KX;
  let fy = KY;
  if (isNum(e.x) && isNum(e.width)) {
    const h = mapSpan(e.x, e.width, mapX, KX);
    out.x = h.start;
    out.width = h.size;
    fx = h.factor;
  } else if (isNum(e.x)) {
    out.x = mapX(e.x);
  }
  if (isNum(e.y) && isNum(e.height)) {
    const v = mapSpan(e.y, e.height, mapY, KY);
    out.y = v.start;
    out.height = v.size;
    fy = v.factor;
  } else if (isNum(e.y)) {
    out.y = mapY(e.y);
  }
  // Same wrap width relative to the letters, so lines break where they did.
  if (e.kind === 'text' && isNum(e.size)) out.size = e.size * fx;
  if (e.kind === 'photo' && isObj(e.crop)) {
    const crop = e.crop;
    // Growing the image by the larger frame factor keeps every in-bounds offset in bounds,
    // so no white strip appears (the renderer does not clamp crops).
    out.crop = {
      ...crop,
      ...(isNum(crop.offsetX) ? { offsetX: crop.offsetX * fx } : {}),
      ...(isNum(crop.offsetY) ? { offsetY: crop.offsetY * fy } : {}),
      zoom: (isNum(crop.zoom) ? crop.zoom : 1) * (Math.max(fx, fy) / Math.min(fx, fy)),
    };
  }
  return out;
}

function scaleKeys(o: Obj, keys: string[], k: number): Obj {
  const out: Obj = {};
  for (const key of keys) if (isNum(o[key])) out[key] = (o[key] as number) * k;
  return out;
}

/**
 * Raw zine data of format version 1 (A5 pages) converted to format 2 (A6 mm), before parseZine
 * reads it. Only geometry changes; missing or broken fields are left for parseZine's defaults.
 * Never changes its input.
 */
export function migrateA5ToA6(raw: Obj): Obj {
  const out: Obj = { ...raw, formatVersion: 2 };
  if (isObj(raw.margins)) {
    const m = raw.margins;
    out.margins = { ...m, ...scaleKeys(m, ['outer', 'inner'], KX), ...scaleKeys(m, ['top', 'bottom'], KY) };
  }
  if (Array.isArray(raw.spreads)) {
    out.spreads = raw.spreads.map((s) =>
      isObj(s) && Array.isArray(s.elements) ? { ...s, elements: s.elements.map(migrateElement) } : s,
    );
  }
  return out;
}
```

```ts
export function parseZine(input: unknown): Zine {
  if (!isObj(input)) throw invalid();
  const version = num(input.formatVersion, 1);
  if (version > FORMAT_VERSION) {
    throw new ZineFormatError('newer', 'Ten zin zapisała nowsza wersja ZINIARZA. Odśwież stronę i spróbuj jeszcze raz.');
  }
  // Format 1 had A5 pages; everything after this line reads A6 mm.
  const data = version < 2 ? migrateA5ToA6(input) : input;
  if (typeof data.id !== 'string' || typeof data.sheets !== 'number' || !Array.isArray(data.spreads)) {
    throw invalid();
  }
  const sheets = clampSheets(data.sheets);
  if (sheets !== data.sheets || data.spreads.length !== spreadCount(sheets)) throw invalid();
```

```ts
import { BLEED, PAGE_HEIGHT, PAGE_WIDTH, SPREAD_WIDTH, type Rect } from '../model/geometry';
import type { PrintSettings } from '../model/types';

export const A4_LANDSCAPE = { width: 297, height: 210 } as const;
/** Space reserved outside the bleed for crop or fold marks. */
export const MARK_BAND = 5;

export type PrintLayoutOptions = Pick<PrintSettings, 'bleed' | 'cropMarks' | 'foldMarks'>;

export interface PrintLayout {
  /** Bleed on the sheet, mm (0 when bleed is off). */
  bleed: number;
  markBand: number;
  /** Trimmed pair of pages (the A5 leaf) on the sheet, mm. */
  spread: Rect;
  trimPage: { width: number; height: number };
}

/**
 * The A5 leaf (two A6 pages) at 100% in the middle of an A4 landscape sheet. More than
 * 20 mm of paper stays around the bleed and marks, so every home printer prints it whole.
 */
export function printLayout(o: PrintLayoutOptions): PrintLayout {
  return {
    bleed: o.bleed ? BLEED : 0,
    markBand: o.cropMarks || o.foldMarks ? MARK_BAND : 0,
    spread: {
      x: (A4_LANDSCAPE.width - SPREAD_WIDTH) / 2,
      y: (A4_LANDSCAPE.height - PAGE_HEIGHT) / 2,
      width: SPREAD_WIDTH,
      height: PAGE_HEIGHT,
    },
    trimPage: { width: PAGE_WIDTH, height: PAGE_HEIGHT },
  };
}
```
